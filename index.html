const express = require("express"); const axios = require("axios"); const cors = require("cors"); require("dotenv").config(); const app = express(); app.use(cors()); app.use(express.json()); const PORT = process.env.PORT || 3000; /* DURANFX / VERTEX FX M-PESA DARaja backend IMPORTANT: Put your Daraja credentials in .env. NEVER put them inside index.html. */ let transactions = []; /* ========================= HEALTH CHECK ========================= */ app.get("/api/health", (req, res) => { res.json({ success: true, service: "Vertex FX payment backend", status: "online" }); }); /* ========================= GET DARAJA ACCESS TOKEN ========================= */ async function getAccessToken() { const credentials = Buffer.from( process.env.MPESA_CONSUMER_KEY + ":" + process.env.MPESA_CONSUMER_SECRET ).toString("base64"); const response = await axios.get( "https://api.safaricom.co.ke/oauth/v1/generate?grant_type=client_credentials", { headers: { Authorization: "Basic " + credentials } } ); return response.data.access_token; } /* ========================= M-PESA STK PUSH ========================= */ app.post("/api/mpesa/deposit", async (req, res) => { try { const { phone, amount, userId } = req.body; if (!phone || !amount || !userId) { return res.status(400).json({ success: false, message: "Phone, amount and userId are required." }); } const token = await getAccessToken(); const timestamp = new Date() .toISOString() .replace(/[-T:.Z]/g, "") .substring(0, 14); const password = Buffer.from( process.env.MPESA_SHORTCODE + process.env.MPESA_PASSKEY + timestamp ).toString("base64"); const payload = { BusinessShortCode: process.env.MPESA_SHORTCODE, Password: password, Timestamp: timestamp, TransactionType: "CustomerPayBillOnline", Amount: Number(amount), PartyA: phone, PartyB: process.env.MPESA_SHORTCODE, PhoneNumber: phone, CallBackURL: process.env.MPESA_CALLBACK_URL, AccountReference: "VERTEXFX", TransactionDesc: "Vertex FX Deposit" }; const response = await axios.post( "https://api.safaricom.co.ke/mpesa/stkpush/v1/processrequest", payload, { headers: { Authorization: "Bearer " + token, "Content-Type": "application/json" } } ); transactions.push({ userId: userId, type: "DEPOSIT", amount: Number(amount), phone: phone, status: "PENDING", checkoutRequestID: response.data.CheckoutRequestID, createdAt: new Date().toISOString() }); res.json({ success: true, message: "M-Pesa payment request sent to the customer's phone.", checkoutRequestID: response.data.CheckoutRequestID }); } catch(error) { console.error( "M-Pesa deposit error:", error.response?.data || error.message ); res.status(500).json({ success: false, message: "Unable to start M-Pesa payment." }); } }); /* ========================= M-PESA CALLBACK ========================= */ app.post("/api/mpesa/callback", (req, res) => { console.log( "M-Pesa callback:", JSON.stringify(req.body, null, 2) ); const callback = req.body?.Body?.stkCallback; if (!callback) { return res.json({ ResultCode: 0, ResultDesc: "Accepted" }); } const checkoutRequestID = callback.CheckoutRequestID; const transaction = transactions.find( item => item.checkoutRequestID === checkoutRequestID ); if (transaction) { if (callback.ResultCode === 0) { transaction.status = "COMPLETED"; transaction.mpesaReceipt = callback.CallbackMetadata?.Item ?.find( item => item.Name === "MpesaReceiptNumber" )?.Value; } else { transaction.status = "FAILED"; transaction.failureReason = callback.ResultDesc; } } res.json({ ResultCode: 0, ResultDesc: "Accepted" }); }); /* ========================= TRANSACTION STATUS ========================= */ app.get( "/api/transactions/:userId", (req, res) => { const userTransactions = transactions.filter( item => String(item.userId) === String(req.params.userId) ); res.json({ success: true, transactions: userTransactions }); } ); /* ========================= WITHDRAWAL REQUEST ========================= */ app.post("/api/mpesa/withdraw", async (req, res) => { try { const { phone, amount, userId } = req.body; if (!phone || !amount || !userId) { return res.status(400).json({ success: false, message: "Phone, amount and userId are required." }); } /* A withdrawal is first recorded as PENDING. Do not automatically send money until: - the user's balance has been checked - the withdrawal is authorized - your M-Pesa B2C setup is approved */ const withdrawal = { id: Date.now(), userId: userId, phone: phone, amount: Number(amount), type: "WITHDRAWAL", status: "PENDING", createdAt: new Date().toISOString() }; transactions.push( withdrawal ); res.json({ success: true, message: "Withdrawal request received.", withdrawalId: withdrawal.id }); } catch(error) { console.error(error); res.status(500).json({ success: false, message: "Withdrawal request failed." }); } }); /* ========================= START SERVER ========================= */ app.listen( PORT, () => { console.log( `Vertex FX backend running on port ${PORT}` ); } );
